import express from "express";
import { puter } from "@heyputer/puter.js";

const app = express();
app.use(express.json());

// Standard homepage web chat interface route
app.get("/", (req, res) => {
    res.send(`Serving Chat Web Interface...`);
});

// NEW ROUTE: Optimized for Discord Bot message callbacks
app.post("/api/discord", async (req, res) => {
    try {
        // Supports standard payload variations "content" or "message" sent by bots
        const userMessage = req.body.content || req.body.message || req.body.prompt;
        
        if (!userMessage) {
            return res.status(400).json({ error: "Missing content in JSON body" });
        }

        // Route prompt to Puter AI using the elite Claude engine model
        const aiResponse = await puter.ai.chat(userMessage, {
            model: "anthropic/claude-opus-4-7"
        });

        let replyText = typeof aiResponse === "string" 
            ? aiResponse 
            : JSON.stringify(aiResponse);

        let isTruncated = false;
        // Critical Guardrail: Discord strict 2,000 character limit enforcement
        if (replyText.length > 2000) {
            replyText = replyText.substring(0, 1995) + "...";
            isTruncated = true;
        }

        // Return the optimized JSON structure Discord APIs love
        res.json({
            content: replyText,
            truncated: isTruncated
        });

    } catch(err) {
        res.status(500).json({
            error: err.message || "Server routing failed"
        });
    }
});

app.listen(3000, () => {
    console.log("running on http://localhost:3000");
});

CRITICAL: Running Puter in Node.js Backend

Puter.js uses automatic browser popups for keyless login in frontend apps. Because a **Node.js server** has no browser, you must explicitly authenticate using an API token when running your bot locally or on GitHub/VPS!

1. Go to Puter.com and sign in.
2. Open the console (F12) on Puter.com, type puter.authToken, and hit Enter.
3. Copy the string token and load it into your backend environment as an environment variable:
PUTER_AUTH_TOKEN=your_copied_token_here

const { Client, GatewayIntentBits } = require('discord.js');
const { init } = require('@heyputer/puter.js/src/init.cjs');
const express = require('express');

// 1. Initialize Puter with your background Token
// Make sure you run your server with: PUTER_AUTH_TOKEN=your_token node index.js
const puter = init(process.env.PUTER_AUTH_TOKEN);

const client = new Client({
    intents: [
        GatewayIntentBits.Guilds,
        GatewayIntentBits.GuildMessages,
        GatewayIntentBits.MessageContent
    ]
});

client.on('messageCreate', async (message) => {
    if (message.author.bot) return;

    if (message.content.startsWith('!ask ')) {
        const question = message.content.slice(5).trim();
        if (!question) return;

        await message.channel.sendTyping();

        try {
            // Ask Claude via the authorized Puter AI module
            const reply = await puter.ai.chat(question, {
                model: "anthropic/claude-opus-4-7"
            });

            const content = typeof reply === 'string' ? reply : reply.message.content.toString();

            // Discord has a strict limit of 2000 characters
            if (content.length > 2000) {
                await message.reply(content.substring(0, 1995) + '...');
            } else {
                await message.reply(content);
            }

        } catch (error) {
            console.error(error);
            await message.reply("⚠️ Puter API Gateway Error. Ensure PUTER_AUTH_TOKEN is correct.");
        }
    }
});

client.login('YOUR_DISCORD_BOT_TOKEN');

import discord
from discord.ext import commands
import requests

intents = discord.Intents.default()
intents.message_content = True

bot = commands.Bot(command_prefix='!', intents=intents)

@bot.command()
async def ask(ctx, *, question: str):
    # Triggers Discord visual typing block
    async with ctx.typing():
        try:
            # Query the Express server endpoint running locally
            api_url = "http://localhost:3000/api/discord"
            response = requests.post(api_url, json={"content": question})
            data = response.json()
            
            # Send the resulting generated answer back to Discord
            if "content" in data:
                await ctx.reply(data["content"])
            else:
                await ctx.send("⚠️ Received empty payload from local API server.")
                
        except Exception as e:
            print(f"Exception during post: {e}")
            await ctx.send("⚠️ Error communicating with Express bot backend.")

bot.run('YOUR_DISCORD_BOT_TOKEN')
